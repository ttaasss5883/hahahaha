                           class="w-full bg-slate-900 border border-slate-700 rounded-lg p-2.5 text-sm focus:outline-none focus:border-purple-500">

    <!-- JavaScriptロジック -->
    <script>
        // 15軸の定義
        const axes = [
            "展開のスピード", "ストーリーの複雑さ", "緊張感", "伏線回収", "狂気度", 
            "世界観の深さ", "ダーク度", "中毒性", "熱量・勢い", "作画の美しさ", 
            "キャラクターの魅力", "感動・泣ける", "独自性・シュール", "爽快感", "考察のしがい"
        ];

        // 最小限のデータベース（あいまい検索用：これにヒットするか、新規なら自動補完されます）
        let animeDatabase = [
            { title: "進撃の巨人", scores: [7,6,7,7,7,6,7,7,7,7,7,6,6,7,7] },
            { title: "呪術廻戦", scores: [6,5,6,5,6,5,6,7,7,7,7,5,5,7,5] },
            { title: "チェンソーマン", scores: [6,5,7,5,7,7,7,7,7,6,7,5,7,6,6] },
            { title: "葬送のフリーレン", scores: [3,6,4,6,2,7,3,6,4,7,7,7,5,4,6] }
        ];

        let coins = 3;
        let radarChartInstance = null;

        // 初期化処理：スライダーを自動生成
        function initSliders() {
            const container = document.getElementById('sliders-container');
            const reviewContainer = document.getElementById('review-sliders-container');
            
            let html = '';
            axes.forEach((axis, i) => {
                html += `
                    <div class="bg-slate-900/60 p-3 rounded-xl border border-slate-800">
                        <div class="flex justify-between text-xs mb-1">
                            <span class="font-medium text-slate-300">${axis}</span>
                            <span id="val-${i}" class="text-purple-400 font-bold">4点</span>
                        </div>
                        <input type="range" min="1" max="7" value="4" id="slider-${i}" 
                               oninput="document.getElementById('val-${i}').innerText = this.value + '点'"
                               class="w-full accent-purple-500 cursor-pointer">
                    </div>
                `;
            });
            container.innerHTML = html;
            reviewContainer.innerHTML = html;
        }

        // Fuse.js あいまい検索の設定
        const fuse = new Fuse(animeDatabase, { keys: ['title'], threshold: 0.4 });

        document.getElementById('user-search-input').addEventListener('input', (e) => {
            const query = e.target.value;
            const feedback = document.getElementById('search-feedback');
            if (!query) {
                feedback.innerText = "💡 入力に合わせて自動で作品を推測します";
                return;
            }
            const results = fuse.search(query);
            if (results.length > 0) {
                feedback.innerText = `🎯 予測マッチ: 「${results[0].item.title}」`;
            } else {
                feedback.innerText = "✨ 新規作品として登録・分析します";
            }
        });

        // 診断実行ロジック
        function runDiagnosis() {
            if (coins < 1) {
                alert("コインが足りません！「作品を評価」タブからコインを獲得してください。");
                switchTab('review');
                return;
            }

            coins -= 1;
            updateCoinDisplay();

            // ユーザーの入力値を取得
            let userScores = [];
            for (let i = 0; i < axes.length; i++) {
                userScores.push(parseInt(document.getElementById(`slider-${i}`).value));
            }

            const query = document.getElementById('user-search-input').value.trim();
            let targetAnime = animeDatabase[0];

            if (query) {
                const results = fuse.search(query);
                if (results.length > 0) {
                    targetAnime = results[0].item;
                } else {
                    // データベースにない場合は簡易的にターゲットを作成
                    targetAnime = { title: query, scores: userScores.map(v => Math.min(7, Math.max(1, v + (Math.random() * 2 - 1)))) };
                }
            }

            // 類似度（シンクロ率）の簡易計算
            let diffSum = 0;
            for (let i = 0; i < axes.length; i++) {
                let diff = Math.abs(userScores[i] - targetAnime.scores[i]);
                diffSum += diff;
            }
            let maxDiff = axes.length * 6;
            let similarity = Math.max(70, Math.round((1 - (diffSum / maxDiff)) * 100 * 10) / 10);

            // 結果表示
            document.getElementById('result-container').classList.remove('hidden');
            document.getElementById('match-title').innerText = targetAnime.title;
            document.getElementById('match-score').innerText = `シンクロ率: ${similarity}%`;

            // レーダーチャート描画
            drawRadarChart(userScores, targetAnime.scores, targetAnime.title);
            
            // スクロール誘導
            document.getElementById('result-container').scrollIntoView({ behavior: 'smooth' });
        }

        // レーダーチャートを描画
        function drawRadarChart(userScores, animeScores, animeTitle) {
            const ctx = document.getElementById('radarChart').getContext('2d');
            if (radarChartInstance) {
                radarChartInstance.destroy();
            }

            radarChartInstance = new Chart(ctx, {
                type: 'radar',
                data: {
                    labels: axes,
                    datasets: [
                        {
                            label: 'あなたの好み',
                            data: userScores,
                            backgroundColor: 'rgba(168, 85, 247, 0.2)',
                            borderColor: 'rgba(168, 85, 247, 1)',
                            borderWidth: 2,
                        },
                        {
                            label: animeTitle,
                            data: animeScores,
                            backgroundColor: 'rgba(236, 72, 153, 0.2)',
                            borderColor: 'rgba(236, 72, 153, 1)',
                            borderWidth: 2,
                        }
                    ]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    scales: {
                        r: {
                            min: 0,
                            max: 7,
                            ticks: { display: false, stepSize: 1 },
                            grid: { color: 'rgba(255, 255, 255, 0.1)' },
                            angleLines: { color: 'rgba(255, 255, 255, 0.1)' },
                            pointLabels: { color: '#cbd5e1', font: { size: 10 } }
                        }
                    },
                    plugins: { legend: { labels: { color: '#fff', font: { size: 11 } } } }
                }
            });
        }

        // コイン獲得（評価送信）
        function submitReview() {
            const animeName = document.getElementById('review-anime-name').value.trim();
            if (!animeName) {
                alert("作品名を入力してください！");
                return;
            }
            coins += 1;
            updateCoinDisplay();
            alert(`「${animeName}」の評価を送信しました！コインが1枚増えました！`);
            document.getElementById('review-anime-name').value = '';
        }

        // コイン表示の更新
        function updateCoinDisplay() {
            document.getElementById('coin-count').innerText = coins;
            document.getElementById('mypage-coins').innerText = coins + "枚";
        }

        // タブ切り替え
        function switchTab(tabId) {
            ['diagnosis', 'review', 'mypage'].forEach(id => {
                document.getElementById(`tab-${id}`).classList.add('hidden');
                document.getElementById(`btn-${id}`).classList.remove('text-purple-400', 'font-bold');
                document.getElementById(`btn-${id}`).classList.add('text-slate-400');
            });
            document.getElementById(`tab-${tabId}`).classList.remove('hidden');
            document.getElementById(`btn-${tabId}`).classList.remove('text-slate-400');
            document.getElementById(`btn-${tabId}`).classList.add('text-purple-400', 'font-bold');
        }

        // 起動時初期化
        window.onload = initSliders;
    </script>
